##고급 타입들

<h3>인터섹션 타입</h3>
type Person = {
    name: string;
    age: number;
};

type Employee = {
employeeId: number;
department: string;
};

type Staff = Person & Employee;
☞ & 연산자로 합쳐 두 타입을 모두 갖는 새로운 타입을 만듦.
☞ 이 교차 타입은 두 타입의 모든 속성을 포함함으로 이 타입으로 객체를 생성할 때 하나의 속성이라도 빠지면 컴파일 오류 발생.
☞ 서로 다른 타입의 속성을 결합해 더 구체적인 타입을 만들 때 유용.

const alice: Staff = {
name: "Alice",
age: 30,
employeeId: 101,
department: "HR",
};
